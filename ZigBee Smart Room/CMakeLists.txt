cmake_minimum_required(VERSION 3.16)
project(zigbee_smart_room_core C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(room_core STATIC firmware/common/measurement.c firmware/common/actuator.c firmware/common/zcl_values.c firmware/coordinator/room_policy.c firmware/device_control/device_control.c)
target_include_directories(room_core PUBLIC firmware/common firmware/coordinator firmware/device_control)
if(MSVC)
  target_compile_options(room_core PRIVATE /W4)
else()
  target_compile_options(room_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(room_core PUBLIC m)
endif()
